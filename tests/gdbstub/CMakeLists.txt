cmake_minimum_required(VERSION 3.16)
project(cemu_gdbstub_tests LANGUAGES C CXX)

option(GDBSTUB_SANITIZERS "Enable address and undefined behavior sanitizers" OFF)
set(CORE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../../core")
file(GLOB_RECURSE CORE_SOURCES CONFIGURE_DEPENDS "${CORE_DIR}/*.c")
list(FILTER CORE_SOURCES EXCLUDE REGEX "/arm/|/os/os-win32\\.c$")
if(WIN32)
    list(FILTER CORE_SOURCES EXCLUDE REGEX "/os/os-linux\\.c$")
    list(APPEND CORE_SOURCES "${CORE_DIR}/os/os-win32.c")
endif()

list(FILTER CORE_SOURCES EXCLUDE REGEX "/debug/gdbstub\\.c$")
add_library(gdb-test-core OBJECT ${CORE_SOURCES})
target_compile_features(gdb-test-core PRIVATE c_std_11)
add_executable(cemu-gdb-headless $<TARGET_OBJECTS:gdb-test-core>
    ${CORE_DIR}/debug/gdbstub.c
    ../autotester/autotester.cpp ../autotester/headless_cli.cpp)
target_compile_definitions(gdb-test-core PRIVATE DEBUG_SUPPORT)
target_compile_definitions(cemu-gdb-headless PRIVATE DEBUG_SUPPORT)
if(UNIX)
    target_compile_definitions(gdb-test-core PRIVATE GLOB_SUPPORT)
    target_compile_definitions(cemu-gdb-headless PRIVATE GLOB_SUPPORT)
endif()
target_compile_features(cemu-gdb-headless PRIVATE c_std_11 cxx_std_11)
if(WIN32)
    target_link_libraries(cemu-gdb-headless PRIVATE ws2_32)
elseif(APPLE)
    target_link_libraries(cemu-gdb-headless PRIVATE
        "-framework CoreFoundation" "-framework IOKit")
endif()
if(GDBSTUB_SANITIZERS)
    target_compile_options(gdb-test-core PRIVATE -fsanitize=address,undefined
        -fno-omit-frame-pointer)
    target_compile_options(cemu-gdb-headless PRIVATE -fsanitize=address,undefined
        -fno-omit-frame-pointer)
    target_link_options(cemu-gdb-headless PRIVATE -fsanitize=address,undefined)
endif()

add_executable(gdb-watch-tests watch_tests.c $<TARGET_OBJECTS:gdb-test-core>)
target_compile_definitions(gdb-watch-tests PRIVATE DEBUG_SUPPORT)
target_compile_features(gdb-watch-tests PRIVATE c_std_11)
if(WIN32)
    target_link_libraries(gdb-watch-tests PRIVATE ws2_32)
elseif(APPLE)
    target_link_libraries(gdb-watch-tests PRIVATE
        "-framework CoreFoundation" "-framework IOKit")
elseif(UNIX)
    target_link_libraries(gdb-watch-tests PRIVATE m)
endif()
if(GDBSTUB_SANITIZERS)
    target_compile_options(gdb-watch-tests PRIVATE -fsanitize=address,undefined
        -fno-omit-frame-pointer)
    target_link_options(gdb-watch-tests PRIVATE -fsanitize=address,undefined)
endif()

add_executable(gdb-state-tests state_tests.c)
target_compile_definitions(gdb-state-tests PRIVATE DEBUG_SUPPORT)
target_compile_features(gdb-state-tests PRIVATE c_std_11)
if(WIN32)
    target_link_libraries(gdb-state-tests PRIVATE ws2_32)
endif()
if(GDBSTUB_SANITIZERS)
    target_compile_options(gdb-state-tests PRIVATE -fsanitize=address,undefined
        -fno-omit-frame-pointer)
    target_link_options(gdb-state-tests PRIVATE -fsanitize=address,undefined)
endif()
enable_testing()
add_test(NAME gdb-state-tests COMMAND gdb-state-tests)
add_test(NAME gdb-watch-tests COMMAND gdb-watch-tests)
